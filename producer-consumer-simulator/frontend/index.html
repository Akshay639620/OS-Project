OS Synchronization Flow & Algorithm

Formal synchronization protocol execution path with real-time active step tracking.

Producer Algorithm
1 Generate Item (Data Creation)
↓
2 sem_wait(&empty) [Check Buffer Full]
↓
3 pthread_mutex_lock(&mutex)
↓
4 [CRITICAL SECTION] Insert Item at buffer[in]
↓
5 pthread_mutex_unlock(&mutex)
↓
6 sem_post(&full) [Notify Consumer]
Consumer Algorithm
1 sem_wait(&full) [Check Buffer Empty]
↓
2 pthread_mutex_lock(&mutex)
↓
3 [CRITICAL SECTION] Remove Item from buffer[out]
↓
4 pthread_mutex_unlock(&mutex)
↓
5 sem_post(&empty) [Notify Producer]
↓
6 Process / Consume Item 
/* POSIX Pthreads & Semaphores Producer-Consumer Synchronization */
#include <pthread.h>
#include <semaphore.h>

#define BUFFER_SIZE 8
int buffer[BUFFER_SIZE];
int in = 0, out = 0;

/* Primitives: Mutex for Mutual Exclusion; Semaphores for Counting Available Slots */
pthread_mutex_t mutex = PTHREAD_MUTEX_INITIALIZER;
sem_t empty_slots;  // Initialized to BUFFER_SIZE
sem_t full_items;   // Initialized to 0

void* producer(void* arg) {
    while (1) {
        int item = produce_item();

        sem_wait(&empty_slots);            // Wait if buffer is full
        pthread_mutex_lock(&mutex);        // Acquire exclusive critical section lock

        /* Critical Section */
        buffer[in] = item;
        in = (in + 1) % BUFFER_SIZE;

        pthread_mutex_unlock(&mutex);      // Release critical section lock
        sem_post(&full_items);             // Increment full count & notify consumer
    }
}

void* consumer(void* arg) {
    while (1) {
        sem_wait(&full_items);             // Wait if buffer is empty
        pthread_mutex_lock(&mutex);        // Acquire exclusive critical section lock

        /* Critical Section */
        int item = buffer[out];
        out = (out + 1) % BUFFER_SIZE;

        pthread_mutex_unlock(&mutex);      // Release critical section lock
        sem_post(&empty_slots);            // Increment empty count & notify producer

        consume_item(item);
    }
}
